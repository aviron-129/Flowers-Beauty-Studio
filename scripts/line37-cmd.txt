$ok = 0; $fail = 0
foreach ($s in $sources) {
  Write-Host "DL $($s.file)..."
  try {
    Invoke-WebRequest -Uri $s.url -OutFile $s.file -Headers $headers -UseBasicParsing -TimeoutSec 60
    $sz = (Get-Item $s.file).Length
    if ($sz -gt 10000) { Write-Host "  OK $sz"; $ok++ } else { Write-Host "  TOO SMALL"; $fail++ }
  } catch { Write-Host "  FAIL: $_"; $fail++ }
}
Write-Host "`nDone: $ok ok, $fail fail"

# Check duplicates by hash
Write-Host "`nDuplicate check:"
Get-ChildItem unique-*.jpg | Get-FileHash | Group-Object Hash | Where-Object { $_.Count -gt 1 } | ForEach-Object {
  Write-Host "DUPLICATE: $($_.Group.Name -join ', ')"
}
