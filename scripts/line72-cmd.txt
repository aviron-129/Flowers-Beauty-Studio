$hashes = @{}
$num = 0

foreach ($s in $sources) {
  if ($s.src -eq "pexels") {
    $url = "https://images.pexels.com/photos/$($s.id)/pexels-photo-$($s.id).jpeg?auto=compress&cs=tinysrgb&w=1200"
  } else {
    $url = "https://images.unsplash.com/$($s.slug)?fm=jpg&q=85&w=1200&auto=format&fit=crop&ixlib=rb-4.0.3"
  }

  $tmp = Join-Path $folder "_tmp.jpg"
  try {
    Invoke-WebRequest -Uri $url -OutFile $tmp -Headers $headers -UseBasicParsing -TimeoutSec 45
    $sz = (Get-Item $tmp).Length
    if ($sz -lt 15000) { Remove-Item $tmp; Write-Host "SKIP small: $($s.id)"; continue }
    $hash = (Get-FileHash $tmp).Hash
    if ($hashes.ContainsKey($hash)) {
      Remove-Item $tmp
      Write-Host "SKIP dup: $($s.id) = same as $($hashes[$hash])"
      continue
    }
    $num++
    $dest = Join-Path $folder ("flower-{0:D2}.jpg" -f $num)
    Move-Item $tmp $dest -Force
    $hashes[$hash] = $s.id
    Write-Host "OK flower-$('{0:D2}' -f $num) <- $($s.id) ($sz bytes)"
  } catch {
    Write-Host "FAIL $($s.id): $_"
    if (Test-Path $tmp) { Remove-Item $tmp }
  }
}

Write-Host "`n=== DONE: $num unique photos in $folder ==="
Get-ChildItem $folder -Filter "flower-*.jpg" | Select-Object Name, Length
